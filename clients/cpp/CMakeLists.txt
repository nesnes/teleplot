cmake_minimum_required(VERSION 3.15)
project(teleplot_cpp LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(Threads REQUIRED)

if(MSVC)
    add_compile_options(/W4 /WX /permissive-)
else()
    add_compile_options(-Wall -Wextra -Wpedantic -Werror)
endif()

# Tests: ./test.sh runs them all (with the checks done by the library of the page)
add_executable(teleplot_test_client tests/test_client.cpp)
add_executable(teleplot_test_scenarios tests/scenarios.cpp)
add_executable(teleplot_test_image tests/test_image.cpp)
target_link_libraries(teleplot_test_image Threads::Threads)
target_link_libraries(teleplot_test_client Threads::Threads)
target_link_libraries(teleplot_test_scenarios Threads::Threads)
if(WIN32)
    target_link_libraries(teleplot_test_client ws2_32)
    target_link_libraries(teleplot_test_scenarios ws2_32)
    target_link_libraries(teleplot_test_image ws2_32)
endif()

# The examples (one idea each, the same set in every language: clients/examples.js): built, to check they compile everywhere
file(GLOB TELEPLOT_EXAMPLES CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/examples/*.cpp")
foreach(example ${TELEPLOT_EXAMPLES})
    get_filename_component(name ${example} NAME_WE)
    add_executable(example_${name} ${example})
    target_link_libraries(example_${name} Threads::Threads)
    if(WIN32)
        target_link_libraries(example_${name} ws2_32)
    endif()
endforeach()
